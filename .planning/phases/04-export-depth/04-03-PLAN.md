// EXP-01/02/03 HTML 내보내기 어댑터 — @dworks/exporters 를 web 환경에서 호출.
// CONTEXT D-04 (download UI 통합).

import { exportTreeToHtml } from '@dworks/exporters'
import type { Tree } from '@dworks/tree'

export interface DownloadHtmlInput {
  tree: Tree
  label: string
}

export function buildHtmlFilename(exportedAt: Date, label: string): string {
  // generation-export.ts 의 buildExportFilename 패턴과 유사
  const stamp = exportedAt.toISOString().replace(/[:.]/g, '-').slice(0, 19)
  const safeLabel = label.replace(/[^a-zA-Z0-9가-힣_-]/g, '-').slice(0, 40)
  return `${stamp}-${safeLabel || 'export'}.html`
}

export function downloadGenerationAsHtml(input: DownloadHtmlInput, exportedAt: Date): { filename: string } {
  const html = exportTreeToHtml(input.tree, { title: input.label, lang: 'ko' })
  const filename = buildHtmlFilename(exportedAt, input.label)
  const blob = new Blob([html], { type: 'text/html;charset=utf-8' })
  const href = URL.createObjectURL(blob)
  try {
    const anchor = document.createElement('a')
    anchor.href = href
    anchor.download = filename
    document.body.appendChild(anchor)
    anchor.click()
    anchor.remove()
  } finally {
    URL.revokeObjectURL(href)
  }
  return { filename }
}
```

function handleExportGenerationHtml(): void {
  const active = generations.find((entry) => entry.id === activeGenerationId)
  if (active === undefined) return
  const exportedAt = new Date()
  try {
    const { filename } = downloadGenerationAsHtml(
      { tree: active.tree, label: active.label },
      exportedAt,
    )
    showToast({
      tone: 'success',
      message: `${filename} 다운로드`,
    })
  } catch (error) {
    showToast({
      tone: 'error',
      message: `HTML 내보내기 실패: ${error instanceof Error ? error.message : '알 수 없는 오류'}`,
    })
  }
}
```

3. 기존 JSON 내보내기 버튼 옆에 HTML 버튼 추가 (generation 패널 자리 검색해 `handleExportGeneration` 호출 버튼 근처):
