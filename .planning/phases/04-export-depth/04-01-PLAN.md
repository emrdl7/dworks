```typescript
function renderCarousel(node: CarouselNode): string {
  const items = node.slides ?? []
  if (items.length === 0) return ''
  const slides = items
    .map((slide, idx) =>
      `<div class="dw-carousel__slide" data-dworks-carousel-slide data-slide-index="${idx}">${renderNode(slide as TreeNode)}</div>`,
    )
    .join('')
  return `<div${commonAttrs(node)} class="dw-carousel" data-dworks-carousel><div class="dw-carousel__viewport"><div class="dw-carousel__container">${slides}</div></div><button class="dw-carousel__btn dw-carousel__btn--prev" data-dworks-carousel-prev aria-label="이전 슬라이드">‹</button><button class="dw-carousel__btn dw-carousel__btn--next" data-dworks-carousel-next aria-label="다음 슬라이드">›</button></div>`
}
```

**renderTabs** (현재 activeIndex panel 만 → 전체 trigger + panel 렌더):
```typescript
function renderTabs(node: TabsNode): string {
  const tabs = node.tabs ?? []
  if (tabs.length === 0) return ''
  const activeId = tabs[node.activeIndex ?? 0]?.id ?? tabs[0]?.id ?? ''
  const triggers = tabs
    .map((tab) =>
      `<button class="dw-tabs__trigger" role="tab" data-dworks-tab-trigger data-tab-id="${escapeHtml(tab.id)}" aria-controls="panel-${escapeHtml(tab.id)}" aria-selected="${tab.id === activeId}" tabindex="${tab.id === activeId ? '0' : '-1'}">${escapeHtml(tab.label ?? '')}</button>`,
    )
    .join('')
  const panels = tabs
    .map((tab) =>
      `<div class="dw-tabs__panel" role="tabpanel" data-dworks-tab-panel data-tab-id="${escapeHtml(tab.id)}" id="panel-${escapeHtml(tab.id)}" aria-labelledby="${escapeHtml(tab.id)}"${tab.id === activeId ? '' : ' hidden'}>${renderChildren((tab.content ?? []) as TreeNode[])}</div>`,
    )
    .join('')
  return `<div${commonAttrs(node)} class="dw-tabs" data-dworks-tabs><div class="dw-tabs__list" role="tablist">${triggers}</div>${panels}</div>`
}
```

**renderAccordion** (`expandMode` data-attr 추가):
```typescript
function renderAccordion(node: AccordionNode): string {
  const items = node.items ?? []
  const expandMode = node.expandMode ?? 'multiple'
  const html = items
    .map((item) =>
      `<details class="dw-accordion__item" data-dworks-accordion-item${item.expanded ? ' open' : ''}><summary class="dw-accordion__summary">${escapeHtml(item.title ?? '')}</summary><div class="dw-accordion__panel">${renderChildren((item.content ?? []) as TreeNode[])}</div></details>`,
    )
    .join('')
  return `<div${commonAttrs(node)} class="dw-accordion" data-dworks-accordion data-expand-mode="${expandMode}">${html}</div>`
}
```

**renderModal** (trigger 버튼 + close 버튼 박기):
```typescript
function renderModal(node: ModalNode): string {
  const triggerLabel = node.triggerLabel ?? '열기'
  const body = renderChildren((node.children ?? []) as TreeNode[])
  return `<button class="dw-modal__trigger" data-dworks-modal-trigger data-modal-id="${escapeHtml(node.id)}">${escapeHtml(triggerLabel)}</button><dialog${commonAttrs(node)} class="dw-modal" data-dworks-modal id="${escapeHtml(node.id)}">${body}<button class="dw-modal__close" data-dworks-modal-close aria-label="닫기">닫기</button></dialog>`
}
```

각 노드의 schema field 이름은 실제 packages/tree/src 정의와 일치하도록 검증. 누락 필드는 fallback (`??`) 으로 처리. 모든 사용자 입력은 `escapeHtml()` 통과.
