```typescript
// EXP-01/02/03 HTML exporter — Embla CDN + vanilla JS + IntersectionObserver wrap.
// CONTEXT D-01 (full document), D-02 (feature detect), D-03 (inline CSS lite), D-04 (download UI hook).

import { renderTree } from '@dworks/tree-renderer'
import type { Tree, TreeNode } from '@dworks/tree'

import {
  TABS_SCRIPT,
  MODAL_SCRIPT,
  ACCORDION_SCRIPT,
  CAROUSEL_INIT_SCRIPT,
  ENTRANCE_ANIMATION_SCRIPT,
  BASE_CSS,
} from './html-scripts'

export interface ExportFeatures {
  hasCarousel: boolean
  hasTabs: boolean
  hasAccordion: boolean
  hasModal: boolean
  hasEntranceAnimation: boolean
}

export interface ExportHtmlOptions {
  title?: string
  lang?: string  // 기본 'ko'
}

const EMBLA_CDN = 'https://unpkg.com/embla-carousel@8.5.0/embla-carousel.umd.js'
const PRETENDARD_CDN = 'https://cdn.jsdelivr.net/gh/orioncactus/pretendard@v1.3.9/dist/web/variable/pretendardvariable.min.css'

export function detectFeatures(tree: Tree): ExportFeatures {
  const features: ExportFeatures = {
    hasCarousel: false,
    hasTabs: false,
    hasAccordion: false,
    hasModal: false,
    hasEntranceAnimation: false,
  }

  function visit(node: TreeNode): void {
    if (node.type === 'carousel') features.hasCarousel = true
    if (node.type === 'tabs') features.hasTabs = true
    if (node.type === 'accordion') features.hasAccordion = true
    if (node.type === 'modal') features.hasModal = true
    if ((node as { entranceAnimation?: unknown }).entranceAnimation !== undefined) {
      features.hasEntranceAnimation = true
    }
    const children = (node as { children?: TreeNode[] }).children
    if (Array.isArray(children)) children.forEach(visit)
    // carousel slides + tabs panels + accordion items + modal children
    const slides = (node as { slides?: TreeNode[] }).slides
    if (Array.isArray(slides)) slides.forEach(visit)
    const tabs = (node as { tabs?: Array<{ content?: TreeNode[] }> }).tabs
    if (Array.isArray(tabs)) tabs.forEach((t) => (t.content ?? []).forEach(visit))
    const items = (node as { items?: Array<{ content?: TreeNode[] }> }).items
    if (Array.isArray(items)) items.forEach((i) => (i.content ?? []).forEach(visit))
  }

  visit(tree.root as TreeNode)
  return features
}

export function exportTreeToHtml(tree: Tree, options: ExportHtmlOptions = {}): string {
  const title = options.title ?? 'Dworks Export'
  const lang = options.lang ?? 'ko'
  const body = renderTree(tree)
  const features = detectFeatures(tree)
  const head = buildHead(title, lang, features)
  const scripts = buildScripts(features)
  return `<!DOCTYPE html>\n<html lang="${escape(lang)}">\n<head>${head}</head>\n<body>\n${body}\n${scripts}\n</body>\n</html>\n`
}

function buildHead(title: string, lang: string, features: ExportFeatures): string {
  const parts: string[] = []
  parts.push(`<meta charset="UTF-8">`)
  parts.push(`<meta name="viewport" content="width=device-width, initial-scale=1.0">`)
  parts.push(`<title>${escape(title)}</title>`)
  parts.push(`<link rel="stylesheet" href="${PRETENDARD_CDN}">`)
  parts.push(`<style>${BASE_CSS}</style>`)
  if (features.hasCarousel) {
    parts.push(`<script src="${EMBLA_CDN}" defer></script>`)
  }
  return parts.join('\n')
}

function buildScripts(features: ExportFeatures): string {
  const scripts: string[] = []
  if (features.hasTabs) scripts.push(`<script>${TABS_SCRIPT}</script>`)
  if (features.hasModal) scripts.push(`<script>${MODAL_SCRIPT}</script>`)
  if (features.hasAccordion) scripts.push(`<script>${ACCORDION_SCRIPT}</script>`)
  if (features.hasCarousel) scripts.push(`<script>${CAROUSEL_INIT_SCRIPT}</script>`)
  if (features.hasEntranceAnimation) scripts.push(`<script>${ENTRANCE_ANIMATION_SCRIPT}</script>`)
  return scripts.join('\n')
}

function escape(s: string): string {
  return s.replace(/[&<>"']/g, (ch) => {
    const map: Record<string, string> = { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }
    return map[ch] ?? ch
  })
}
```

```typescript
// EXP-02 vanilla JS templates + EXP-03 IntersectionObserver.
// 각 template 은 export 시 <script> 태그 안에 그대로 inject 됨. IIFE 패턴.

export const TABS_SCRIPT = `
(function(){
  document.querySelectorAll('[data-dworks-tabs]').forEach(function(root){
    var triggers = root.querySelectorAll('[data-dworks-tab-trigger]');
    var panels = root.querySelectorAll('[data-dworks-tab-panel]');
    function activate(tabId){
      triggers.forEach(function(t){
        var active = t.getAttribute('data-tab-id') === tabId;
        t.setAttribute('aria-selected', active);
        t.setAttribute('tabindex', active ? '0' : '-1');
      });
      panels.forEach(function(p){
        var active = p.getAttribute('data-tab-id') === tabId;
        if (active) p.removeAttribute('hidden');
        else p.setAttribute('hidden', '');
      });
    }
    triggers.forEach(function(t, i){
      t.addEventListener('click', function(){ activate(t.getAttribute('data-tab-id')); });
      t.addEventListener('keydown', function(e){
        if (e.key === 'ArrowRight') { var n = triggers[(i+1) % triggers.length]; n.focus(); activate(n.getAttribute('data-tab-id')); }
        if (e.key === 'ArrowLeft') { var p = triggers[(i-1+triggers.length) % triggers.length]; p.focus(); activate(p.getAttribute('data-tab-id')); }
      });
    });
  });
})();
`

export const MODAL_SCRIPT = `
(function(){
  document.querySelectorAll('[data-dworks-modal-trigger]').forEach(function(btn){
    btn.addEventListener('click', function(){
      var id = btn.getAttribute('data-modal-id');
      var dialog = document.getElementById(id);
      if (dialog && typeof dialog.showModal === 'function') dialog.showModal();
    });
  });
  document.querySelectorAll('[data-dworks-modal-close]').forEach(function(btn){
    btn.addEventListener('click', function(){
      var dialog = btn.closest('dialog');
      if (dialog) dialog.close();
    });
  });
  // backdrop 클릭 close
  document.querySelectorAll('[data-dworks-modal]').forEach(function(dialog){
    dialog.addEventListener('click', function(e){
      var rect = dialog.getBoundingClientRect();
      var inside = e.clientX >= rect.left && e.clientX <= rect.right && e.clientY >= rect.top && e.clientY <= rect.bottom;
      if (!inside) dialog.close();
    });
  });
})();
`

export const ACCORDION_SCRIPT = `
(function(){
  document.querySelectorAll('[data-dworks-accordion]').forEach(function(root){
    if (root.getAttribute('data-expand-mode') !== 'single') return;
    var items = root.querySelectorAll('[data-dworks-accordion-item]');
    items.forEach(function(item){
      item.addEventListener('toggle', function(){
        if (item.open) {
          items.forEach(function(other){ if (other !== item) other.removeAttribute('open'); });
        }
      });
    });
  });
})();
`

export const CAROUSEL_INIT_SCRIPT = `
(function(){
  function init(){
    if (typeof EmblaCarousel !== 'function') return;
    document.querySelectorAll('[data-dworks-carousel]').forEach(function(root){
      var viewport = root.querySelector('.dw-carousel__viewport');
      if (!viewport) return;
      var embla = EmblaCarousel(viewport, { loop: true, slidesToScroll: 1 });
      var prev = root.querySelector('[data-dworks-carousel-prev]');
      var next = root.querySelector('[data-dworks-carousel-next]');
      if (prev) prev.addEventListener('click', function(){ embla.scrollPrev(); });
      if (next) next.addEventListener('click', function(){ embla.scrollNext(); });
    });
  }
  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', init);
  } else {
    init();
  }
})();
`

export const ENTRANCE_ANIMATION_SCRIPT = `
(function(){
  var prefersReduced = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  var TYPE_TO_INITIAL = {
    'slide-up': 'translateY(20px)',
    'slide-left': 'translateX(20px)',
    'slide-right': 'translateX(-20px)',
    'slide-down': 'translateY(-20px)',
    'fade-in': null
  };
  function applyInitial(el){
    var type = el.getAttribute('data-entrance-animation') || 'fade-in';
    if (prefersReduced) { el.style.opacity = '1'; el.style.transform = 'none'; return; }
    el.style.opacity = '0';
    var t = TYPE_TO_INITIAL[type];
    if (t) el.style.transform = t;
  }
  function applyFinal(el){
    var duration = el.getAttribute('data-entrance-duration') || '600';
    var delay = el.getAttribute('data-entrance-delay') || '0';
    el.style.transition = 'opacity ' + duration + 'ms ease ' + delay + 'ms, transform ' + duration + 'ms ease ' + delay + 'ms';
    el.style.opacity = '1';
    el.style.transform = 'none';
  }
  var nodes = document.querySelectorAll('[data-entrance-animation]');
  if (prefersReduced) { nodes.forEach(function(el){ applyInitial(el); }); return; }
  var loadEls = [];
  var scrollEls = [];
  nodes.forEach(function(el){
    var trigger = el.getAttribute('data-entrance-trigger') || 'scroll';
    applyInitial(el);
    if (trigger === 'load') loadEls.push(el);
    else scrollEls.push(el);
  });
  // load trigger
  window.addEventListener('DOMContentLoaded', function(){
    requestAnimationFrame(function(){ loadEls.forEach(applyFinal); });
  });
  // scroll trigger via IntersectionObserver
  if ('IntersectionObserver' in window) {
    var observer = new IntersectionObserver(function(entries){
      entries.forEach(function(entry){
        if (entry.isIntersecting) { applyFinal(entry.target); observer.unobserve(entry.target); }
      });
    }, { threshold: 0.1 });
    scrollEls.forEach(function(el){ observer.observe(el); });
  } else {
    scrollEls.forEach(applyFinal);
  }
})();
`
```

```typescript
import { test } from 'node:test'
import assert from 'node:assert/strict'

import { detectFeatures, exportTreeToHtml } from './html'

function makeTree(rootChildren: unknown[]): { root: { id: string; type: string; children: unknown[] } } {
  return { root: { id: 'root', type: 'section', children: rootChildren } } as never
}

test('detectFeatures — 인터랙티브 4종 + entranceAnimation 검출', () => {
  const tree = makeTree([
    { id: 'c1', type: 'carousel', slides: [] },
    { id: 't1', type: 'tabs', tabs: [] },
    { id: 'a1', type: 'accordion', items: [] },
    { id: 'm1', type: 'modal', children: [] },
    { id: 'x1', type: 'text', entranceAnimation: { type: 'slide-up' } },
  ])
  const features = detectFeatures(tree as never)
  assert.equal(features.hasCarousel, true)
  assert.equal(features.hasTabs, true)
  assert.equal(features.hasAccordion, true)
  assert.equal(features.hasModal, true)
  assert.equal(features.hasEntranceAnimation, true)
})

test('exportTreeToHtml — !DOCTYPE + lang ko + head meta', () => {
  const tree = makeTree([])
  const html = exportTreeToHtml(tree as never)
  assert.ok(html.startsWith('<!DOCTYPE html>'))
  assert.ok(html.includes('lang="ko"'))
  assert.ok(html.includes('<meta charset="UTF-8">'))
  assert.ok(html.includes('pretendardvariable'))
})

test('exportTreeToHtml — carousel 있으면 Embla CDN + init script 주입', () => {
  const tree = makeTree([{ id: 'c1', type: 'carousel', slides: [] }])
  const html = exportTreeToHtml(tree as never)
  assert.ok(html.includes('embla-carousel@8.5.0'))
  assert.ok(html.includes('EmblaCarousel('))
})

test('exportTreeToHtml — 인터랙티브 없으면 script 미주입', () => {
  const tree = makeTree([{ id: 't', type: 'text', text: 'hi' }])
  const html = exportTreeToHtml(tree as never)
  assert.ok(!html.includes('EmblaCarousel'))
  assert.ok(!html.includes('data-dworks-tab-trigger'))
})

test('exportTreeToHtml — entranceAnimation 있으면 IntersectionObserver 주입', () => {
  const tree = makeTree([{ id: 'x', type: 'text', entranceAnimation: { type: 'slide-up' } }])
  const html = exportTreeToHtml(tree as never)
  assert.ok(html.includes('IntersectionObserver'))
  assert.ok(html.includes('prefers-reduced-motion'))
})
```
