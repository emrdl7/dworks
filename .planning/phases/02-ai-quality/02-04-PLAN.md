    ```typescript
      tree: Tree,
    ```

    **A) generation-normalize.ts** — Task 1 의 ensureInteractiveComponent stub 자리에 실제 구현:

    ```typescript
    const DOMAIN_TO_INTERACTIVE: Record<DomainKey, 'carousel' | 'tabs' | 'accordion' | 'modal'> = {
      ecommerce: 'carousel',
      saas: 'tabs',
      faq: 'accordion',
      auth: 'modal',
      default: 'accordion',
    }

    /** main 컨테이너 (role === 'main') 를 찾아 반환. 없으면 undefined. */
    function findMainContainer(node: TreeNode): TreeNode | undefined {
      const n = node as unknown as Record<string, unknown>
      if (n.role === 'main' && 'children' in node && Array.isArray((node as { children?: unknown }).children)) {
        return node
      }
      if ('children' in node && Array.isArray((node as { children?: TreeNode[] }).children)) {
        for (const child of (node as { children: TreeNode[] }).children) {
          const found = findMainContainer(child)
          if (found !== undefined) return found
        }
      }
      return undefined
    }

    /** TreeNode 가 interactive type 인지 검사. */
    function isInteractive(node: TreeNode): boolean {
      return node.type === 'carousel' || node.type === 'tabs' || node.type === 'accordion' || node.type === 'modal'
    }

    /** 트리 어디든 interactive 노드가 1개 이상 존재하는지. */
    function hasAnyInteractive(node: TreeNode): boolean {
      if (isInteractive(node)) return true
      if ('children' in node && Array.isArray((node as { children?: TreeNode[] }).children)) {
        for (const child of (node as { children: TreeNode[] }).children) {
          if (hasAnyInteractive(child)) return true
        }
      }
      return false
    }

    /** 빈 interactive 노드 fabricate (placeholder text 포함). */
    function makeInteractiveNode(type: 'carousel' | 'tabs' | 'accordion' | 'modal'): TreeNode {
      const id = `auto-${type}-${Date.now().toString(36)}`
      const placeholder = '인터랙티브 요소 — 도메인에 맞게 편집하세요'
      // schema 호환: carousel/tabs/accordion 은 items, modal 은 children + label.
      if (type === 'modal') {
        return {
          id,
          type: 'modal',
          editKind: 'structure',
          label: placeholder,
          children: [],
        } as unknown as TreeNode
      }
      // carousel / tabs / accordion — items 배열 1 슬롯
      return {
        id,
        type,
        editKind: 'structure',
        items: [
          {
            id: `${id}-slot-1`,
            label: placeholder,
            children: [],
          },
        ],
      } as unknown as TreeNode
    }

    /** main 직계 children 의 마지막 section/hero 직전에 노드 삽입. D-06. */
    function insertBeforeLastSection(mainNode: TreeNode, newNode: TreeNode): TreeNode {
      if (!('children' in mainNode) || !Array.isArray((mainNode as { children?: TreeNode[] }).children)) {
        return mainNode
      }
      const children = (mainNode as { children: TreeNode[] }).children
      // 마지막 section/hero index
      let lastIdx = -1
      for (let i = children.length - 1; i >= 0; i--) {
        const c = children[i]
        if (c !== undefined && (c.type === 'section' || c.type === 'hero')) {
          lastIdx = i
          break
        }
      }
      const insertAt = lastIdx === -1 ? children.length : lastIdx
      const nextChildren = [...children.slice(0, insertAt), newNode, ...children.slice(insertAt)]
      return { ...mainNode, children: nextChildren } as TreeNode
    }

    /** 재귀로 트리 안의 특정 노드를 교체. */
    function replaceNode(node: TreeNode, targetId: string, replacement: TreeNode): TreeNode {
      if (node.id === targetId) return replacement
      if ('children' in node && Array.isArray((node as { children?: TreeNode[] }).children)) {
        const children = (node as { children: TreeNode[] }).children
        return {
          ...node,
          children: children.map((c) => replaceNode(c, targetId, replacement)),
        } as TreeNode
      }
      return node
    }

    function ensureInteractiveComponent(tree: Tree, domain: DomainKey): Tree {
      // D-07: 이미 interactive 있으면 skip (idempotent safety)
      if (hasAnyInteractive(tree.root)) return tree
      const interactiveType = DOMAIN_TO_INTERACTIVE[domain]
      const newNode = makeInteractiveNode(interactiveType)
      const main = findMainContainer(tree.root)
      if (main === undefined) {
        // main 없으면 root.children 직접 — fallback (root 가 container 일 때만)
        if ('children' in tree.root && Array.isArray((tree.root as { children?: TreeNode[] }).children)) {
          const replaced = insertBeforeLastSection(tree.root, newNode)
          return { ...tree, root: replaced }
        }
        return tree
      }
      const replacedMain = insertBeforeLastSection(main, newNode)
      return { ...tree, root: replaceNode(tree.root, main.id, replacedMain) }
    }
    ```

    **A) generation-normalize.ts** — Task 1 의 ensureEntranceAnimations stub 자리에 실제 구현:

    ```typescript
    interface EntranceAnimation {
      type: 'slide-up' | 'fade-in' | 'slide-left' | 'slide-right' | 'scale-up'
      duration: number
      delay?: number
      trigger: 'load' | 'scroll'
    }

    const MAX_ANIMATIONS = 8
    const STAGGER_DELAY_MS = 150

    /** TreeNode 의 entranceAnimation prop 이 이미 있는지. */
    function hasEntranceAnimation(node: TreeNode): boolean {
      const n = node as unknown as Record<string, unknown>
      return n.entranceAnimation !== undefined && n.entranceAnimation !== null
    }

    /** 노드에 entranceAnimation 부착 — 기존 값 있으면 그대로 둠 (D-10). */
    function withEntranceAnimation(node: TreeNode, anim: EntranceAnimation): TreeNode {
      if (hasEntranceAnimation(node)) return node
      return { ...node, entranceAnimation: anim } as unknown as TreeNode
    }

    /** heading 자식 찾기 — emphasis === 'heading-1' 또는 type === 'text' 의 첫 자식. */
    function findHeadingChild(container: TreeNode): TreeNode | undefined {
      if (!('children' in container) || !Array.isArray((container as { children?: TreeNode[] }).children)) {
        return undefined
      }
      const children = (container as { children: TreeNode[] }).children
      // 우선 emphasis='heading-1' 매칭, 없으면 첫 text
      for (const c of children) {
        const n = c as unknown as Record<string, unknown>
        if (n.emphasis === 'heading-1') return c
      }
      for (const c of children) {
        if (c.type === 'text') return c
      }
      return undefined
    }

    /** main 직계 section/hero 들 반환. */
    function findMainDirectSections(tree: Tree): TreeNode[] {
      const main = findMainContainer(tree.root)
      if (main === undefined) return []
      if (!('children' in main) || !Array.isArray((main as { children?: TreeNode[] }).children)) return []
      return (main as { children: TreeNode[] }).children.filter(
        (c) => c.type === 'section' || c.type === 'hero',
      )
    }

    /** 트리 전체 재귀로 특정 id 의 노드를 변환 함수로 교체. */
    function transformNode(node: TreeNode, targetId: string, transform: (n: TreeNode) => TreeNode): TreeNode {
      if (node.id === targetId) return transform(node)
      if ('children' in node && Array.isArray((node as { children?: TreeNode[] }).children)) {
        const children = (node as { children: TreeNode[] }).children
        return {
          ...node,
          children: children.map((c) => transformNode(c, targetId, transform)),
        } as TreeNode
      }
      return node
    }

    function ensureEntranceAnimations(tree: Tree): Tree {
      let placed = 0
      let next = tree
      const mainSections = findMainDirectSections(tree)
      if (mainSections.length === 0) return tree

      // 우선순위 1: 첫 hero 의 heading → slide-up 700 trigger:load
      const firstHero = mainSections.find((s) => s.type === 'hero')
      if (firstHero !== undefined) {
        const heroHeading = findHeadingChild(firstHero)
        if (heroHeading !== undefined && !hasEntranceAnimation(heroHeading)) {
          next = { ...next, root: transformNode(next.root, heroHeading.id, (n) =>
            withEntranceAnimation(n, { type: 'slide-up', duration: 700, trigger: 'load' })) }
          placed += 1
        }
      }

      // 우선순위 2: main 직계 section/hero 의 _직접 자식_ 카드들 staggered slide-up 600 scroll
      // (첫 hero 자체의 자식은 skip — 우선순위 1 의 heading 과 충돌 회피)
      for (const sec of mainSections) {
        if (placed >= MAX_ANIMATIONS) break
        // sec 가 첫 hero 면 skip (위에서 처리한 heading 외 다른 자식들 staggered)
        if (!('children' in sec) || !Array.isArray((sec as { children?: TreeNode[] }).children)) continue
        const sectionChildren = (sec as { children: TreeNode[] }).children
        let staggerIdx = 0
        for (const card of sectionChildren) {
          if (placed >= MAX_ANIMATIONS) break
          if (hasEntranceAnimation(card)) continue
          // 첫 hero 의 heading 자식이면 skip (이미 처리)
          if (sec === firstHero && card.type === 'text') continue
          const delay = staggerIdx * STAGGER_DELAY_MS
          next = { ...next, root: transformNode(next.root, card.id, (n) =>
            withEntranceAnimation(n, { type: 'slide-up', duration: 600, delay, trigger: 'scroll' })) }
          placed += 1
          staggerIdx += 1
        }
      }

      // 우선순위 3: 마지막 section/hero 의 heading/button → fade-in 500 scroll
      if (placed < MAX_ANIMATIONS) {
        const last = mainSections[mainSections.length - 1]
        if (last !== undefined) {
          const lastHeading = findHeadingChild(last)
          if (lastHeading !== undefined && !hasEntranceAnimation(lastHeading) && placed < MAX_ANIMATIONS) {
            next = { ...next, root: transformNode(next.root, lastHeading.id, (n) =>
              withEntranceAnimation(n, { type: 'fade-in', duration: 500, trigger: 'scroll' })) }
            placed += 1
          }
        }
      }

      return next
    }
    ```
