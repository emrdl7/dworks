```typescript
'use client'
// DS-01 theme switcher — 3 테마 (dark | light | sepia) chrome 토큰 swap.
// CONTEXT D-04 (localStorage dw:theme:v1), D-05 (ThemeProvider + useTheme hook).

import {
  createContext,
  useCallback,
  useContext,
  useEffect,
  useMemo,
  useState,
  type ReactNode,
} from 'react'

export type Theme = 'dark' | 'light' | 'sepia'

const THEME_STORAGE_KEY = 'dw:theme:v1'
const VALID_THEMES: Theme[] = ['dark', 'light', 'sepia']

interface ThemeContextValue {
  theme: Theme
  setTheme: (next: Theme) => void
}

const ThemeContext = createContext<ThemeContextValue | null>(null)

function readSavedTheme(): Theme {
  if (typeof window === 'undefined') return 'dark'
  try {
    const saved = window.localStorage.getItem(THEME_STORAGE_KEY)
    if (saved !== null && VALID_THEMES.includes(saved as Theme)) {
      return saved as Theme
    }
  } catch {
    // localStorage 차단 — 기본 dark
  }
  return 'dark'
}

export function ThemeProvider({ children }: { children: ReactNode }) {
  const [theme, setThemeState] = useState<Theme>('dark')

  // hydration — 클라이언트 첫 마운트 시 localStorage 읽고 attr 적용
  useEffect(() => {
    const saved = readSavedTheme()
    setThemeState(saved)
    document.documentElement.setAttribute('data-theme', saved)
  }, [])

  const setTheme = useCallback((next: Theme) => {
    setThemeState(next)
    document.documentElement.setAttribute('data-theme', next)
    try {
      window.localStorage.setItem(THEME_STORAGE_KEY, next)
    } catch {
      // 차단 — 메모리만 갱신
    }
  }, [])

  const value = useMemo<ThemeContextValue>(
    () => ({ theme, setTheme }),
    [theme, setTheme],
  )

  return <ThemeContext.Provider value={value}>{children}</ThemeContext.Provider>
}

export function useTheme(): ThemeContextValue {
  const ctx = useContext(ThemeContext)
  if (ctx === null) {
    throw new Error('useTheme must be used within ThemeProvider')
  }
  return ctx
}
```
