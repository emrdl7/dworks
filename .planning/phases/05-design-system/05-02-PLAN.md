```typescript
'use client'
// DS-02 custom palette — 사용자 정의 색 set 저장/관리.
// CONTEXT D-06 (5-10색), D-07 (localStorage dw:palettes:v1), D-09 (Provider + hook).

import {
  createContext,
  useCallback,
  useContext,
  useEffect,
  useMemo,
  useState,
  type ReactNode,
} from 'react'

export interface Palette {
  id: string
  name: string
  colors: string[]  // hex 문자열 배열, 5-10개
}

const STORAGE_KEY = 'dw:palettes:v1'
const MAX_PALETTES = 8
const MIN_COLORS_PER_PALETTE = 5
const MAX_COLORS_PER_PALETTE = 10

interface PaletteContextValue {
  palettes: Palette[]
  activePaletteId: string | null
  addPalette: (palette: Omit<Palette, 'id'>) => Palette | null
  removePalette: (id: string) => void
  updatePalette: (id: string, patch: Partial<Omit<Palette, 'id'>>) => void
  setActivePaletteId: (id: string | null) => void
}

const PaletteContext = createContext<PaletteContextValue | null>(null)

function isValidHex(value: string): boolean {
  return /^#[0-9a-fA-F]{3}([0-9a-fA-F]{3})?$/.test(value)
}

function isValidPalette(value: unknown): value is Palette {
  if (typeof value !== 'object' || value === null) return false
  const v = value as Record<string, unknown>
  if (typeof v.id !== 'string') return false
  if (typeof v.name !== 'string') return false
  if (!Array.isArray(v.colors)) return false
  if (v.colors.length < MIN_COLORS_PER_PALETTE || v.colors.length > MAX_COLORS_PER_PALETTE) return false
  return v.colors.every((c) => typeof c === 'string' && isValidHex(c))
}

function readSaved(): { palettes: Palette[]; activeId: string | null } {
  if (typeof window === 'undefined') return { palettes: [], activeId: null }
  try {
    const raw = window.localStorage.getItem(STORAGE_KEY)
    if (raw === null) return { palettes: [], activeId: null }
    const parsed = JSON.parse(raw) as { palettes?: unknown; activeId?: unknown }
    const palettes = Array.isArray(parsed.palettes) ? parsed.palettes.filter(isValidPalette) : []
    const activeId = typeof parsed.activeId === 'string' ? parsed.activeId : null
    return { palettes, activeId }
  } catch {
    return { palettes: [], activeId: null }
  }
}

function persist(palettes: Palette[], activeId: string | null): void {
  if (typeof window === 'undefined') return
  try {
    window.localStorage.setItem(STORAGE_KEY, JSON.stringify({ palettes, activeId }))
  } catch {
    // 차단 — 메모리만
  }
}

function generateId(): string {
  return `pal-${Date.now().toString(36)}-${Math.random().toString(36).slice(2, 7)}`
}

export function PaletteProvider({ children }: { children: ReactNode }) {
  const [palettes, setPalettes] = useState<Palette[]>([])
  const [activePaletteId, setActivePaletteIdState] = useState<string | null>(null)

  useEffect(() => {
    const saved = readSaved()
    setPalettes(saved.palettes)
    setActivePaletteIdState(saved.activeId)
  }, [])

  const addPalette = useCallback((partial: Omit<Palette, 'id'>): Palette | null => {
    if (palettes.length >= MAX_PALETTES) return null
    if (partial.colors.length < MIN_COLORS_PER_PALETTE) return null
    if (partial.colors.length > MAX_COLORS_PER_PALETTE) return null
    if (!partial.colors.every(isValidHex)) return null
    const next: Palette = { id: generateId(), ...partial }
    const list = [...palettes, next]
    setPalettes(list)
    persist(list, activePaletteId)
    return next
  }, [palettes, activePaletteId])

  const removePalette = useCallback((id: string) => {
    const list = palettes.filter((p) => p.id !== id)
    const nextActive = activePaletteId === id ? null : activePaletteId
    setPalettes(list)
    setActivePaletteIdState(nextActive)
    persist(list, nextActive)
  }, [palettes, activePaletteId])

  const updatePalette = useCallback((id: string, patch: Partial<Omit<Palette, 'id'>>) => {
    const list = palettes.map((p) => {
      if (p.id !== id) return p
      const next: Palette = { ...p, ...patch }
      if (patch.colors !== undefined) {
        if (next.colors.length < MIN_COLORS_PER_PALETTE || next.colors.length > MAX_COLORS_PER_PALETTE) return p
        if (!next.colors.every(isValidHex)) return p
      }
      return next
    })
    setPalettes(list)
    persist(list, activePaletteId)
  }, [palettes, activePaletteId])

  const setActivePaletteId = useCallback((id: string | null) => {
    setActivePaletteIdState(id)
    persist(palettes, id)
  }, [palettes])

  const value = useMemo<PaletteContextValue>(
    () => ({ palettes, activePaletteId, addPalette, removePalette, updatePalette, setActivePaletteId }),
    [palettes, activePaletteId, addPalette, removePalette, updatePalette, setActivePaletteId],
  )

  return <PaletteContext.Provider value={value}>{children}</PaletteContext.Provider>
}

export function usePalettes(): PaletteContextValue {
  const ctx = useContext(PaletteContext)
  if (ctx === null) {
    throw new Error('usePalettes must be used within PaletteProvider')
  }
  return ctx
}

export {
  MAX_PALETTES,
  MIN_COLORS_PER_PALETTE,
  MAX_COLORS_PER_PALETTE,
}
```

```tsx
'use client'
// DS-02 palette manager dialog — 사용자 정의 palette CRUD UI.
// CONTEXT D-08 (인스펙터 통합은 deferred), D-10 (dialog UI + 색 picker grid).

import { useState } from 'react'

import {
  MAX_COLORS_PER_PALETTE,
  MAX_PALETTES,
  MIN_COLORS_PER_PALETTE,
  usePalettes,
  type Palette,
} from './palette-context'
import { useToast } from './toast-context'

export interface PaletteManagerDialogProps {
  open: boolean
  onClose: () => void
}

const DEFAULT_COLORS: string[] = [
  '#6366f1', '#8b5cf6', '#ec4899', '#ef4444', '#f59e0b',
]

export function PaletteManagerDialog({ open, onClose }: PaletteManagerDialogProps) {
  const { palettes, addPalette, removePalette, updatePalette } = usePalettes()
  const { showToast } = useToast()
  const [draftName, setDraftName] = useState('새 palette')
  const [draftColors, setDraftColors] = useState<string[]>(DEFAULT_COLORS)
  const [editingId, setEditingId] = useState<string | null>(null)

  if (!open) return null

  function handleAdd() {
    if (palettes.length >= MAX_PALETTES) {
      showToast(`최대 ${MAX_PALETTES}개의 palette 만 저장할 수 있습니다`, 'warn')
      return
    }
    if (draftColors.length < MIN_COLORS_PER_PALETTE) {
      showToast(`palette 는 최소 ${MIN_COLORS_PER_PALETTE}색 필요`, 'warn')
      return
    }
    const added = addPalette({ name: draftName.trim() || '이름 없음', colors: draftColors })
    if (added === null) {
      showToast('palette 추가 실패 — 유효성 검사 위반', 'error')
      return
    }
    setDraftName('새 palette')
    setDraftColors(DEFAULT_COLORS)
    showToast(`"${added.name}" palette 추가됨`, 'success')
  }

  function handleColorChange(idx: number, value: string) {
    setDraftColors((prev) => prev.map((c, i) => (i === idx ? value : c)))
  }

  function handleAddColor() {
    if (draftColors.length >= MAX_COLORS_PER_PALETTE) return
    setDraftColors((prev) => [...prev, '#000000'])
  }

  function handleRemoveColor(idx: number) {
    if (draftColors.length <= MIN_COLORS_PER_PALETTE) return
    setDraftColors((prev) => prev.filter((_, i) => i !== idx))
  }

  function handleRemove(p: Palette) {
    removePalette(p.id)
    showToast(`"${p.name}" palette 삭제됨`, 'info')
  }

  return (
    <div
      className="fixed inset-0 z-[1000] flex items-center justify-center bg-black/50"
      onClick={onClose}
      role="dialog"
      aria-modal="true"
      aria-label="팔레트 관리"
    >
      <div
        className="w-[640px] max-h-[80vh] overflow-y-auto rounded-lg bg-[var(--dw-ui-panel)] p-6 shadow-xl"
        onClick={(e) => e.stopPropagation()}
      >
        <div className="mb-4 flex items-center justify-between">
          <h2 className="text-base font-semibold text-[var(--dw-ui-text-primary)]">팔레트 관리</h2>
          <button
            type="button"
            onClick={onClose}
            className="rounded p-1 text-[var(--dw-ui-text-muted)] hover:bg-[var(--dw-ui-soft)]"
            aria-label="닫기"
          >
            ✕
          </button>
        </div>

        {/* 기존 palette 목록 */}
        <div className="mb-6">
          <h3 className="mb-2 text-xs font-medium text-[var(--dw-ui-text-muted)]">
            내 palette ({palettes.length}/{MAX_PALETTES})
          </h3>
          {palettes.length === 0 && (
            <p className="text-xs text-[var(--dw-ui-text-faint)]">아직 저장된 palette 가 없습니다</p>
          )}
          {palettes.map((p) => (
            <div key={p.id} className="mb-2 flex items-center gap-2 rounded border border-[var(--dw-ui-border)] p-2">
              <input
                type="text"
                value={p.name}
                onChange={(e) => updatePalette(p.id, { name: e.target.value })}
                className="w-32 rounded bg-transparent px-2 py-1 text-xs text-[var(--dw-ui-text-primary)]"
                aria-label={`${p.name} 이름`}
              />
              <div className="flex gap-1">
                {p.colors.map((c, i) => (
                  <div
                    key={`${p.id}-${i}`}
                    className="h-6 w-6 rounded border border-[var(--dw-ui-border)]"
                    style={{ backgroundColor: c }}
                    title={c}
                  />
                ))}
              </div>
              <button
                type="button"
                onClick={() => handleRemove(p)}
                className="ml-auto rounded px-2 py-1 text-xs text-[var(--dw-ui-danger-text)] hover:bg-[var(--dw-ui-danger-bg)]"
                aria-label={`${p.name} palette 삭제`}
              >
                삭제
              </button>
            </div>
          ))}
        </div>

        {/* 새 palette 추가 */}
        <div className="rounded border border-[var(--dw-ui-border)] p-3">
          <h3 className="mb-2 text-xs font-medium text-[var(--dw-ui-text-muted)]">새 palette</h3>
          <input
            type="text"
            value={draftName}
            onChange={(e) => setDraftName(e.target.value)}
            placeholder="palette 이름"
            className="mb-3 w-full rounded border border-[var(--dw-ui-border)] bg-[var(--dw-ui-bg)] px-2 py-1 text-xs"
          />
          <div className="mb-3 flex flex-wrap gap-2">
            {draftColors.map((c, idx) => (
              <div key={idx} className="flex items-center gap-1">
                <input
                  type="color"
                  value={c}
                  onChange={(e) => handleColorChange(idx, e.target.value)}
                  className="h-8 w-8 cursor-pointer rounded border border-[var(--dw-ui-border)]"
                  aria-label={`색 ${idx + 1}`}
                />
                {draftColors.length > MIN_COLORS_PER_PALETTE && (
                  <button
                    type="button"
                    onClick={() => handleRemoveColor(idx)}
                    className="h-4 w-4 rounded-full text-[10px] text-[var(--dw-ui-danger-text)] hover:bg-[var(--dw-ui-danger-bg)]"
                    aria-label={`색 ${idx + 1} 제거`}
                  >
                    ✕
                  </button>
                )}
              </div>
            ))}
            {draftColors.length < MAX_COLORS_PER_PALETTE && (
              <button
                type="button"
                onClick={handleAddColor}
                className="h-8 w-8 rounded border border-dashed border-[var(--dw-ui-border)] text-[var(--dw-ui-text-muted)] hover:bg-[var(--dw-ui-soft)]"
                aria-label="색 추가"
              >
                +
              </button>
            )}
          </div>
          <button
            type="button"
            onClick={handleAdd}
            disabled={palettes.length >= MAX_PALETTES}
            className="rounded bg-[var(--dw-ui-accent)] px-3 py-1.5 text-xs font-medium text-white hover:bg-[var(--dw-ui-accent-hover)] disabled:opacity-50"
          >
            palette 추가
          </button>
        </div>
      </div>
    </div>
  )
}
```
