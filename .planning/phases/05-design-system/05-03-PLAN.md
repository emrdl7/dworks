// DS-03 디자인 토큰 export — 트리에서 사용 hex/spacing/radius/fontSize/fontFamily 추출.
// CONTEXT D-11 (extractDesignTokens), D-12 (Tailwind config + CSS variables), D-13 (adapter).

import type { Tree, TreeNode } from '@dworks/tree'

export interface ExtractedTokens {
  colors: string[]       // hex (정렬 + dedup)
  spacing: number[]      // px (정렬 + dedup)
  radius: number[]       // px
  fontSize: number[]     // px
  fontFamily: string[]
}

const HEX_KEYS = ['backgroundColor', 'textColor', 'borderColor', 'accentColor', 'overlayColor']

export function extractDesignTokens(tree: Tree): ExtractedTokens {
  const colors = new Set<string>()
  const spacing = new Set<number>()
  const radius = new Set<number>()
  const fontSize = new Set<number>()
  const fontFamily = new Set<string>()

  function visit(node: TreeNode): void {
    const color = (node as { color?: Record<string, unknown> }).color
    if (color !== undefined) {
      for (const key of HEX_KEYS) {
        const v = color[key]
        if (typeof v === 'string' && /^#[0-9a-fA-F]{3,8}$/.test(v)) {
          colors.add(v.toLowerCase())
        }
      }
    }
    const shape = (node as { shape?: { radius?: number } }).shape
    if (shape !== undefined && typeof shape.radius === 'number') {
      radius.add(shape.radius)
    }
    const spacingProp = (node as { spacing?: Record<string, unknown> }).spacing
    if (spacingProp !== undefined) {
      for (const key of ['padding', 'paddingX', 'paddingY', 'gap', 'margin', 'marginX', 'marginY']) {
        const v = spacingProp[key]
        if (typeof v === 'number') spacing.add(v)
      }
    }
    const typography = (node as { typography?: Record<string, unknown> }).typography
    if (typography !== undefined) {
      const fs = typography.fontSize
      if (typeof fs === 'number') fontSize.add(fs)
      const ff = typography.fontFamily
      if (typeof ff === 'string' && ff.trim().length > 0) fontFamily.add(ff)
    }
    const children = (node as { children?: TreeNode[] }).children
    if (Array.isArray(children)) children.forEach(visit)
  }

  visit(tree.root as TreeNode)

  return {
    colors: Array.from(colors).sort(),
    spacing: Array.from(spacing).sort((a, b) => a - b),
    radius: Array.from(radius).sort((a, b) => a - b),
    fontSize: Array.from(fontSize).sort((a, b) => a - b),
    fontFamily: Array.from(fontFamily).sort(),
  }
}

export type TokenExportFormat = 'tailwind' | 'css'

export function exportDesignTokens(tree: Tree, format: TokenExportFormat): string {
  const tokens = extractDesignTokens(tree)
  if (format === 'tailwind') return formatTailwind(tokens)
  return formatCss(tokens)
}

function formatTailwind(tokens: ExtractedTokens): string {
  const lines: string[] = []
  lines.push('// Dworks 추출 디자인 토큰 — Tailwind config (.ts)')
  lines.push('// /* DS-03 token export */')
  lines.push('import type { Config } from \'tailwindcss\'')
  lines.push('')
  lines.push('const config: Config = {')
  lines.push('  theme: {')
  lines.push('    extend: {')

  if (tokens.colors.length > 0) {
    lines.push('      colors: {')
    tokens.colors.forEach((c, i) => {
      lines.push(`        'dw-${i + 1}': '${c}',`)
    })
    lines.push('      },')
  }

  if (tokens.spacing.length > 0) {
    lines.push('      spacing: {')
    tokens.spacing.forEach((px) => {
      lines.push(`        'dw-${px}': '${px}px',`)
    })
    lines.push('      },')
  }

  if (tokens.radius.length > 0) {
    lines.push('      borderRadius: {')
    tokens.radius.forEach((px) => {
      lines.push(`        'dw-${px}': '${px}px',`)
    })
    lines.push('      },')
  }

  if (tokens.fontSize.length > 0) {
    lines.push('      fontSize: {')
    tokens.fontSize.forEach((px) => {
      lines.push(`        'dw-${px}': '${px}px',`)
    })
    lines.push('      },')
  }

  if (tokens.fontFamily.length > 0) {
    lines.push('      fontFamily: {')
    tokens.fontFamily.forEach((ff, i) => {
      lines.push(`        'dw-${i + 1}': [${JSON.stringify(ff)}],`)
    })
    lines.push('      },')
  }

  lines.push('    },')
  lines.push('  },')
  lines.push('  content: [],')
  lines.push('}')
  lines.push('')
  lines.push('export default config')
  return lines.join('\n') + '\n'
}

function formatCss(tokens: ExtractedTokens): string {
  const lines: string[] = []
  lines.push('/* Dworks 추출 디자인 토큰 — CSS variables */')
  lines.push('/* DS-03 token export */')
  lines.push(':root {')

  tokens.colors.forEach((c, i) => {
    lines.push(`  --dw-token-color-${i + 1}: ${c};`)
  })
  tokens.spacing.forEach((px) => {
    lines.push(`  --dw-token-space-${px}: ${px}px;`)
  })
  tokens.radius.forEach((px) => {
    lines.push(`  --dw-token-radius-${px}: ${px}px;`)
  })
  tokens.fontSize.forEach((px) => {
    lines.push(`  --dw-token-fontsize-${px}: ${px}px;`)
  })
  tokens.fontFamily.forEach((ff, i) => {
    lines.push(`  --dw-token-font-${i + 1}: ${ff};`)
  })

  lines.push('}')
  return lines.join('\n') + '\n'
}

import { exportDesignTokens, extractDesignTokens } from './tokens'

function makeTree(rootChildren: unknown[]): { root: { id: string; type: string; children: unknown[] } } {
  return {
    root: {
      id: 'root',
      type: 'section',
      children: rootChildren,
    },
  } as never
}

test('extractDesignTokens — hex 색 추출 + 정렬', () => {
  const tree = makeTree([
    { id: 't1', type: 'text', color: { backgroundColor: '#FF0000', textColor: '#00FF00' } },
    { id: 't2', type: 'text', color: { borderColor: '#0000ff' } },
  ])
  const tokens = extractDesignTokens(tree as never)
  assert.deepEqual(tokens.colors, ['#0000ff', '#00ff00', '#ff0000'])
})

test('extractDesignTokens — spacing/radius/fontSize 추출 dedup', () => {
  const tree = makeTree([
    { id: 't1', type: 'text', spacing: { padding: 16, gap: 16 }, shape: { radius: 8 }, typography: { fontSize: 48 } },
    { id: 't2', type: 'text', spacing: { padding: 24 }, shape: { radius: 8 } },
  ])
  const tokens = extractDesignTokens(tree as never)
  assert.deepEqual(tokens.spacing, [16, 24])
  assert.deepEqual(tokens.radius, [8])
  assert.deepEqual(tokens.fontSize, [48])
})

test('exportDesignTokens tailwind — config 형식', () => {
  const tree = makeTree([
    { id: 't1', type: 'text', color: { backgroundColor: '#ff6600' }, typography: { fontSize: 24 } },
  ])
  const out = exportDesignTokens(tree as never, 'tailwind')
  assert.ok(out.includes('import type { Config }'))
  assert.ok(out.includes("'dw-1': '#ff6600'"))
  assert.ok(out.includes("'dw-24': '24px'"))
  assert.ok(out.includes('export default config'))
})

test('exportDesignTokens css — variables 형식', () => {
  const tree = makeTree([
    { id: 't1', type: 'text', color: { backgroundColor: '#ff6600' }, spacing: { padding: 16 } },
  ])
  const out = exportDesignTokens(tree as never, 'css')
  assert.ok(out.includes(':root {'))
  assert.ok(out.includes('--dw-token-color-1: #ff6600;'))
  assert.ok(out.includes('--dw-token-space-16: 16px;'))
})

test('exportDesignTokens — 빈 트리는 빈 config/css 생성', () => {
  const tree = makeTree([])
  const tw = exportDesignTokens(tree as never, 'tailwind')
  assert.ok(tw.includes('export default config'))
  const css = exportDesignTokens(tree as never, 'css')
  assert.ok(css.includes(':root {'))
  assert.ok(css.endsWith('}\n'))
})

// DS-03 토큰 export 어댑터 — @dworks/exporters 호출 + Blob download.

import { exportDesignTokens, type TokenExportFormat } from '@dworks/exporters'
import type { Tree } from '@dworks/tree'

export interface DownloadTokensInput {
  tree: Tree
  label: string
}

function buildTokensFilename(exportedAt: Date, label: string, format: TokenExportFormat): string {
  const stamp = exportedAt.toISOString().replace(/[:.]/g, '-').slice(0, 19)
  const safeLabel = label.replace(/[^a-zA-Z0-9가-힣_-]/g, '-').slice(0, 40)
  const ext = format === 'tailwind' ? 'ts' : 'css'
  return `${stamp}-${safeLabel || 'tokens'}.${ext}`
}

export function downloadTokens(
  input: DownloadTokensInput,
  format: TokenExportFormat,
  exportedAt: Date,
): { filename: string } {
  const content = exportDesignTokens(input.tree, format)
  const filename = buildTokensFilename(exportedAt, input.label, format)
  const mime = format === 'tailwind' ? 'text/typescript;charset=utf-8' : 'text/css;charset=utf-8'
  const blob = new Blob([content], { type: mime })
  const href = URL.createObjectURL(blob)
  try {
    const anchor = document.createElement('a')
    anchor.href = href
    anchor.download = filename
    document.body.appendChild(anchor)
    anchor.click()
    anchor.remove()
  } finally {
    URL.revokeObjectURL(href)
  }
  return { filename }
}
