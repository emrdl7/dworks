       export function findNodesInBox(
         boxClientRect: { left: number; top: number; right: number; bottom: number },
         excludeIds: ReadonlySet<string> = new Set(),
       ): string[] {
         const nodes = document.querySelectorAll<HTMLElement>('[data-dworks-node-id]')
         const selected: string[] = []
         nodes.forEach((el) => {
           const id = el.dataset.dworksNodeId
           if (!id || excludeIds.has(id)) return
           const r = el.getBoundingClientRect()
           if (
             r.right > boxClientRect.left &&
             r.left < boxClientRect.right &&
             r.bottom > boxClientRect.top &&
             r.top < boxClientRect.bottom
           ) {
             selected.push(id)
           }
         })
         return selected
       }

         if (!boxSelect.active) return

         function onMove(event: MouseEvent) {
           const rect = el.getBoundingClientRect()
           const z = canvasZoom > 0 ? canvasZoom : 1
           const logicalX = (event.clientX - rect.left + el.scrollLeft) / z
           const logicalY = (event.clientY - rect.top + el.scrollTop) / z

       const distributeH = useCallback(() => {
         if (selectedIds.size < 3) {
           onWarn('분포에는 3개 이상의 노드 선택이 필요합니다.')
           return
         }
         const groups = groupByParent(tree, selectedIds)
         let nextTree = tree
         let focusId: string | null = null
         for (const [parent, children] of groups) {
           if (children.length < 3) continue  // 부모별 3개 미만이면 skip (그룹 단위)
           const nonAbsolute = children.find(c => !isAbsolute(c))
           if (nonAbsolute) {
             onWarn('정렬은 절대 위치(position: absolute) 노드에만 적용됩니다.')
             continue
           }
           // X 축 정렬 후 leftmost/rightmost 사이 균등 분포
           const sorted = [...children].map(c => ({
             node: c,
             rect: getNodeBoundingClientRect(c.id, ...),
           })).sort((a, b) => a.rect.logicalX - b.rect.logicalX)
           const first = sorted[0]
           const last = sorted[sorted.length - 1]
           const span = (last.rect.logicalX) - (first.rect.logicalX)
           const step = span / (sorted.length - 1)
           // first/last 고정, 중간만 균등 배치
           for (let i = 1; i < sorted.length - 1; i++) {
             const newLeft = first.rect.logicalX + step * i
             nextTree = setNodeInset(nextTree, sorted[i].node.id, { left: newLeft })
             focusId = focusId ?? sorted[i].node.id
