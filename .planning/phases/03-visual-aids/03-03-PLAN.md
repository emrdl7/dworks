    구조:
    ```typescript
    'use client'
    // VIS-03 — 좌하단 fixed 미니맵 (D-08..D-12). 200×~120px, 큰 트리 navigate 용.
    // 노드 mini-render 없음 — bounding box outline 만 (성능 회피, D-09).
    // viewport rect: canvasScrollRef.scrollLeft/Top + clientWidth/Height + canvasZoom 으로 계산.
    // 클릭: zoom 변경 없는 navigate → canvasScrollRef.current 의 scrollLeft/Top 직접 설정 (D-11).

    import { useCallback, useEffect, useRef, useState, type RefObject } from 'react'
    import { X } from 'lucide-react'

    const MINIMAP_WIDTH = 200
    const MINIMAP_BODY_HEIGHT_MAX = 120
    const MINIMAP_BODY_HEIGHT_MIN = 60
    const HEADER_HEIGHT = 24
    const VIEWPORT_RECT_BG = 'rgba(99, 102, 241, 0.18)'
    const VIEWPORT_RECT_BORDER = 'rgba(99, 102, 241, 0.6)'
    const NODE_BOX_BORDER = 'rgba(99, 102, 241, 0.35)'

    export interface MinimapProps {
      open: boolean
      canvasScrollRef: RefObject<HTMLElement | null>
      canvasZoom: number
      containerWidth: number
      treeRevision: number | string
      onClose: () => void
    }

    interface NodeBox {
      key: string
      x: number      // logical px
      y: number
      w: number
      h: number
    }

    interface ViewportBox {
      x: number
      y: number
      w: number
      h: number
    }

    export function Minimap(props: MinimapProps): JSX.Element | null {
      const { open, canvasScrollRef, canvasZoom, containerWidth, treeRevision, onClose } = props
      const [nodeBoxes, setNodeBoxes] = useState<NodeBox[]>([])
      const [viewport, setViewport] = useState<ViewportBox | null>(null)
      const [contentSize, setContentSize] = useState<{ w: number; h: number } | null>(null)
      const rafRef = useRef<number | null>(null)

      // 노드 박스 + 콘텐츠 사이즈 측정 (tree 변경 시).
      const measureNodes = useCallback((): void => {
        const wrapper = canvasScrollRef.current
        if (!wrapper) return
        const z = canvasZoom > 0 ? canvasZoom : 1
        const wrapperRect = wrapper.getBoundingClientRect()
        const scrollLeft = wrapper.scrollLeft
        const scrollTop = wrapper.scrollTop
        const all = Array.from(document.querySelectorAll<HTMLElement>('[data-dworks-node-id]'))
        const boxes: NodeBox[] = all.map((el) => {
          const r = el.getBoundingClientRect()
          return {
            key: el.dataset.dworksNodeId ?? `idx-${Math.random()}`,
            x: (r.left - wrapperRect.left + scrollLeft) / z,
            y: (r.top - wrapperRect.top + scrollTop) / z,
            w: r.width / z,
            h: r.height / z,
          }
        })
        setNodeBoxes(boxes)
        // 콘텐츠 사이즈: scrollWidth/Height / zoom (또는 max box right/bottom)
        const cw = wrapper.scrollWidth / z
        const ch = wrapper.scrollHeight / z
        setContentSize({ w: Math.max(cw, containerWidth), h: Math.max(ch, 1) })
      }, [canvasScrollRef, canvasZoom, containerWidth])

      // viewport rect 측정 — scroll/resize/zoom 시.
      const measureViewport = useCallback((): void => {
        const wrapper = canvasScrollRef.current
        if (!wrapper) return
        const z = canvasZoom > 0 ? canvasZoom : 1
        setViewport({
          x: wrapper.scrollLeft / z,
          y: wrapper.scrollTop / z,
          w: wrapper.clientWidth / z,
          h: wrapper.clientHeight / z,
        })
      }, [canvasScrollRef, canvasZoom])

      // 노드 측정 — open 시 + tree 변경 시.
      useEffect(() => {
        if (!open) return
        measureNodes()
        measureViewport()
      }, [open, treeRevision, canvasZoom, containerWidth, measureNodes, measureViewport])

      // scroll/resize listener — rAF throttle.
      useEffect(() => {
        if (!open) return
        const wrapper = canvasScrollRef.current
        if (!wrapper) return
        function onScrollOrResize(): void {
          if (rafRef.current !== null) return
          rafRef.current = window.requestAnimationFrame(() => {
            rafRef.current = null
            measureViewport()
          })
        }
        wrapper.addEventListener('scroll', onScrollOrResize, { passive: true })
        window.addEventListener('resize', onScrollOrResize, { passive: true })
        return () => {
          wrapper.removeEventListener('scroll', onScrollOrResize)
          window.removeEventListener('resize', onScrollOrResize)
          if (rafRef.current !== null) {
            window.cancelAnimationFrame(rafRef.current)
            rafRef.current = null
          }
        }
      }, [open, canvasScrollRef, measureViewport])

      if (!open || !contentSize) {
        // 패널 chrome 은 contentSize 없어도 표시 — null 만 본문 처리
        // 첫 frame 에서 contentSize 계산 전이라도 패널은 보여줘야 사용자가 정상 인지
      }

      const bodyWidth = MINIMAP_WIDTH - 16  // 양옆 8px padding
      const safeContent = contentSize ?? { w: containerWidth, h: containerWidth }
      const aspectH = bodyWidth * (safeContent.h / safeContent.w)
      const bodyHeight = Math.max(
        MINIMAP_BODY_HEIGHT_MIN,
        Math.min(MINIMAP_BODY_HEIGHT_MAX, aspectH),
      )
      const scale = bodyWidth / safeContent.w

      function handleBodyClick(event: React.MouseEvent<HTMLDivElement>): void {
        const wrapper = canvasScrollRef.current
        if (!wrapper) return
        const target = event.currentTarget
        const rect = target.getBoundingClientRect()
        const minimapX = event.clientX - rect.left
        const minimapY = event.clientY - rect.top
        const logicalX = minimapX / scale
        const logicalY = minimapY / scale
        // 클릭 위치를 viewport 중앙 으로 — D-11
        const z = canvasZoom > 0 ? canvasZoom : 1
        wrapper.scrollLeft = logicalX * z - wrapper.clientWidth / 2
        wrapper.scrollTop = logicalY * z - wrapper.clientHeight / 2
        measureViewport()
      }

      if (!open) return null

      return (
        <aside
          role="region"
          aria-label="미니맵"
          className="fixed bottom-4 left-4 overflow-hidden rounded-md border border-[var(--dw-ui-border)] bg-[var(--dw-ui-panel)] shadow-[var(--dw-shadow-elev-2)]"
          style={{ width: MINIMAP_WIDTH, zIndex: 60 }}
        >
          <div
            className="flex items-center justify-between border-b border-[var(--dw-ui-border)] px-2 py-1 text-[11px] font-semibold text-[var(--dw-ui-text-muted)]"
            style={{ height: HEADER_HEIGHT }}
          >
            <span>미니맵</span>
            <button
              type="button"
              aria-label="미니맵 닫기"
              onClick={onClose}
              className="inline-flex h-5 w-5 items-center justify-center rounded text-[var(--dw-ui-text-muted)] transition-colors hover:bg-[var(--dw-ui-elevated)] hover:text-[var(--dw-ui-text-body)] focus-visible:outline focus-visible:outline-2 focus-visible:outline-offset-1 focus-visible:outline-[var(--dw-ui-accent)]"
            >
              <X size={11} aria-hidden="true" />
            </button>
          </div>
          <div
            role="button"
            aria-label="미니맵에서 캔버스 위치 이동"
            tabIndex={0}
            onClick={handleBodyClick}
            onKeyDown={(e) => {
              if (e.key === 'Enter' || e.key === ' ') {
                e.preventDefault()
                // 키보드 활성화 시 중앙으로 (간단 처리)
                const wrapper = canvasScrollRef.current
                if (wrapper) {
                  wrapper.scrollLeft = (safeContent.w * (canvasZoom > 0 ? canvasZoom : 1)) / 2 - wrapper.clientWidth / 2
                  wrapper.scrollTop = (safeContent.h * (canvasZoom > 0 ? canvasZoom : 1)) / 2 - wrapper.clientHeight / 2
                  measureViewport()
                }
              }
            }}
            className="relative cursor-crosshair bg-[var(--dw-ui-canvas-bg)]"
            style={{ width: bodyWidth + 16, height: bodyHeight + 8, padding: '4px 8px' }}
          >
            <div className="relative" style={{ width: bodyWidth, height: bodyHeight }}>
              {nodeBoxes.map((b) => (
                <div
                  key={b.key}
                  className="absolute"
                  style={{
                    left: b.x * scale,
                    top: b.y * scale,
                    width: Math.max(b.w * scale, 1),
                    height: Math.max(b.h * scale, 1),
                    border: `1px solid ${NODE_BOX_BORDER}`,
                    pointerEvents: 'none',
                  }}
                />
              ))}
              {viewport && (
                <div
                  className="absolute"
                  style={{
                    left: viewport.x * scale,
                    top: viewport.y * scale,
                    width: Math.max(viewport.w * scale, 4),
                    height: Math.max(viewport.h * scale, 4),
                    backgroundColor: VIEWPORT_RECT_BG,
                    border: `1px solid ${VIEWPORT_RECT_BORDER}`,
                    pointerEvents: 'none',
                  }}
                />
              )}
            </div>
          </div>
        </aside>
      )
    }
    ```
