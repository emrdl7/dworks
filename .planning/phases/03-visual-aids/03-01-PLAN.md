    ```typescript
    // VIS-01 D-04 — column edges (viewport-aware, logical px).
    function computeColumnEdges(containerWidth: number, columnCount = 12, gutterPx = 16): number[] {
      const colWidth = (containerWidth - (columnCount - 1) * gutterPx) / columnCount
      const edges: number[] = []
      for (let i = 0; i <= columnCount; i += 1) {
        edges.push(i * (colWidth + gutterPx))
        // 마지막 column 의 우측 edge 는 i * (colWidth+gutterPx) - gutterPx 가 정확하지만,
        // snap 목표는 column "시작점" + "끝점" 양쪽 — 시작점은 위 공식, 끝점은 + colWidth
        if (i < columnCount) edges.push(i * (colWidth + gutterPx) + colWidth)
      }
      return Array.from(new Set(edges)).sort((a, b) => a - b)
    }
    function snapToColumnEdge(value: number, edges: number[], threshold = 4): { value: number; snapped: boolean; edge: number | null } {
      let closest = { delta: Infinity, edge: 0 }
      for (const e of edges) {
        const d = Math.abs(value - e)
        if (d < closest.delta) closest = { delta: d, edge: e }
      }
      if (closest.delta <= threshold) return { value: closest.edge, snapped: true, edge: closest.edge }
      return { value, snapped: false, edge: null }
    }
    ```
